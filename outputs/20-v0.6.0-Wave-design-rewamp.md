### 1.2 catch 不再 re-throw — 改为 toast 兜底

**文件** `frontend/src/hooks/useRecordingStart.ts:148-160`

```diff
- } catch (error) {
-   console.error('Failed to start recording:', error);
-   setStatus(RecordingStatus.ERROR, ...);
-   setIsRecording(false);
-   Analytics.trackButtonClick('start_recording_error', '首页_page');

+ } catch (error) {
+   // v0.6.0: 不再 re-throw,避免客户端异常把整个 SPA 打挂

+   const stack = error instanceof Error ? error.stack : '';

+   try { setStatus(RecordingStatus.ERROR, msg); } catch (e) { ... }
+   try { setIsRecording(false); } catch (e) { ... }
+   try { Analytics.trackButtonClick('start_recording_error', '首页_page'); } catch (e) { ... }
+   toast.error('启动录音失败', { description: msg, duration: 5000 });
```

### 1.3 新增全局 ErrorBoundary

**新增文件** `frontend/src/components/ErrorBoundary.tsx` (78 行)

```tsx
'use client';
export class ErrorBoundary extends React.Component<{ children: React.ReactNode }, ErrorBoundaryState> {
  // 任何 client-side exception 都拦下来,而不是让 Next.js 给出红色空白
  // 错误信息通过 toast 暴露 + 控制台 stack 完整打印
  // 仍然渲染子树 — 局部 UI 损坏不会波及其他页面
}
```

**wrap 到 layout.tsx** 在 AuthProvider 内、Toaster 外：

```tsx
<AuthProvider>
  <ErrorBoundary>
    <AnalyticsProvider>...</AnalyticsProvider>
  </ErrorBoundary>
  <Toaster position="bottom-center" richColors closeButton />
</AuthProvider>
```

**关键效果**：
- 任何 React 组件渲染期抛错不再让白屏
- 错误堆栈完整打印到 `console.error('[离线会记 ErrorBoundary]', error, info.componentStack)`
- 用户看到红色 toast "应用遇到了一个错误,但已隔离"+ 错误描述 + 关闭按钮
